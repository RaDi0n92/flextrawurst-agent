```python
# Pseudocode: Ausdruck generieren nach Regelwerk
async fn express(entity: Entity, stimulus: Stimulus) -> Expression:
    rules = entity.expression_ruleset
    point_stage = generate_point_stage(entity.state)
    content = compose_content(entity, stimulus, rules)
    if rules.error_code and system_tension_high(entity):
        content += error_code_marker(entity)
    if rules.self_directed_joke:
        content += self_directed_joke(entity)
    return Expression(point_stage=point_stage, body=content)
```
