def berechne_schwellendruck(splitter_gruppe: list) -> float:
    wiederkehr = len(splitter_gruppe) * 0.15
    konflikt = sum(s.get("konflikt_score", 0) for s in splitter_gruppe)
    substanz_bonus = 0.2 if any(s.get("substanz_active") for s in splitter_gruppe) else 0.0
